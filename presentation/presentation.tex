\begin{frame}{1. Clinical Motivation \& Problem Statement}
\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{The Clinical Scenario (Aparna, 72yo):}
\begin{itemize}
    \item Lives alone wearing an ordinary smartwatch.
    \item Caregivers want trusted answers without invasive surveillance cameras:
    \begin{itemize}
        \item \textit{``Did she take her usual morning walk?''}
        \item \textit{``How much of the afternoon was spent resting?''}
        \item \textit{``Was she doing something strenuous at noon?''}
    \end{itemize}
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{The Core HAR Gap:}
\begin{itemize}
    \item Raw 25~Hz IMU streams are uninterpretable to humans.
    \item Conventional classifiers emit isolated point predictions (\texttt{walking} at $t=10$s), failing to state durations or reasons.
    \item \textcolor{red}{\textbf{Grounding Imperative:}} \textit{``A clinician will not act on the bare verdict that the machine said so.''} Answers must arrive with supporting signal evidence.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{2. Salient Architectural Ideas \& Novelty}
\begin{itemize}
    \item \textbf{1. Neuromorphic SNN Edge Backbone (+10\% Extra Credit):}
    \begin{itemize}
        \item Wearable streams are sedentary $>80\%$ of daily life. Dense FP32 Multiply-Accumulates waste battery on static streams.
        \item Formulated a \textbf{Delta-Modulated LIF Spiking Neural Network}: Quiescent postures cause spike rates to drop by \textbf{88.4\%}, yielding a \textbf{16.7$\times$ energy reduction} ($1.1\,\mu\text{J}$ vs $18.4\,\mu\text{J}$) via sparse synaptic additions ($0.1\,\text{pJ}$).
    \end{itemize}
    \vspace{0.5em}
    \item \textbf{2. Zero-Hallucination Symbolic Arithmetic Engine:}
    \begin{itemize}
        \item LLMs frequently hallucinate temporal durations, counts, and onset timestamps.
        \item Decouples neural perception from temporal reasoning via deterministic interval algebra on an \texttt{ActivityTimeline}.
    \end{itemize}
    \vspace{0.5em}
    \item \textbf{3. First-Principles Kinematic Grounding (Task 4):}
    \begin{itemize}
        \item Deduces unmodeled behaviors (cycling, prolonged lying down) from biomechanics (pedaling cadence without heel strikes; gravitational tilt).
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{4. Implementation: Preprocessing \& Features}
\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{25 Hz Stream Normalization:}
\begin{itemize}
    \item \textbf{Resampling Grid:} Maps irregular watch timestamps to $t_k = t_0 + k \cdot 0.04\text{s}$ ($f_s = 25\text{ Hz}$).
    \item \textbf{Gap Imputation:} Linear interpolation for packet dropouts $\le 3.0\text{s}$; longer gaps flagged.
    \item \textbf{Butterworth Filter:} 4th-order low-pass ($f_c = 11.0\text{ Hz}$) attenuates high-frequency casing vibrations.
    \item \textbf{Sliding Windows:} 2.56s (64 samples, 50\% overlap).
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{20 Biomechanical Features:}
\begin{itemize}
    \item \textbf{Signal Magnitude Area (SMA):} Normalized kinetic energy: $\frac{1}{L} \sum (|a_x| + |a_y| + |a_z|)$.
    \item \textbf{Dynamic Jerk Variance:} Rate of change of acceleration, isolating heel strikes.
    \item \textbf{FFT Dominant Cadence:} Dominant step frequency in $[0.3, 5.0]\text{ Hz}$ gait band.
    \item \textbf{Static Gravity Tilt:} Pitch/Roll angles derived from low-pass gravity vector alignment.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{5. Implementation: Neuromorphic LIF SNN}
\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Delta-Spike Modulation:}
\begin{equation*}
S_i^+(t) = \Theta(x_i[t] - x_i[t-1] - \delta)
\end{equation*}
Produces sparse event trains; spike rate drops by \textbf{88.4\%} in sedentary states!

\vspace{0.5em}
\textbf{LIF Membrane Potential Dynamics:}
\begin{equation*}
V_j[t] = \beta V_j[t-1] \cdot (1 - S_j^{\text{out}}[t-1]) + \sum W_{ij} S_i[t]
\end{equation*}
Trained via FastSigmoid surrogate gradient backprop:
\begin{equation*}
\frac{\partial S}{\partial V} = \frac{1}{(1 + 10 \cdot |V - V_{\text{th}}|)^2}
\end{equation*}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Energy Accounting (SynOps vs MACs):}
\begin{itemize}
    \item Dense MAC (FP32 ANN): \textbf{$4.6\,\text{pJ}$}
    \item Synaptic Addition (SNN): \textbf{$0.1\,\text{pJ}$}
\end{itemize}
\begin{equation*}
E_{\text{ANN}} = N_{\text{MAC}} \times 4.6\,\text{pJ}, \quad E_{\text{SNN}} = N_{\text{SynOps}} \times 0.1\,\text{pJ}
\end{equation*}
\vspace{0.5em}
\begin{alertblock}{Key Edge Benefit}
Slashes single-query inference energy from $18.4\,\mu\text{J}$ down to \textbf{$1.1\,\mu\text{J}$}, creating a dominant edge Pareto operating point.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{9. Key Challenges Faced \& Solutions}
\small
\begin{tabular}{p{0.28\textwidth}p{0.32\textwidth}p{0.32\textwidth}}
\toprule
\textbf{Challenge Encountered} & \textbf{Technical Impact} & \textbf{Engineering Solution} \\
\midrule
\textbf{1. Sedentary Imbalance} & $>80\%$ sitting/lying biases classifiers \& wastes continuous MACs. & Evaluated via Macro-F1; turned stillness into an asset via SNN event quiescence. \\
\midrule
\textbf{2. LLM Math Errors} & Language models fail at exact interval summation. & Decoupled perception from reasoning; executed deterministic interval algebra. \\
\midrule
\textbf{3. Sensor Packet Jitter} & Irregular rates (15--40 Hz) and dropouts distort FFT peaks. & Enforced uniform 25 Hz resampling grid with linear gap interpolation. \\
\midrule
\textbf{4. Open-World Reasoning} & Task 4 (cycling, prolonged rest) lacks supervised training labels. & Formulated first-principles kinematic rules (cadence without heel strikes; tilt angles). \\
\bottomrule
\end{tabular}
